""" App de Disciplina — Flask + SQLite Tarefas: distância (km), leitura (cronômetro), hábito (check). Cada tarefa tem um link único (/t/). Grave esse link numa tag NFC (app "NFC Tools") ou imprima o QR code. Aproximou/escaneou = tarefa concluída. Rodar: pip install flask qrcode[pil] e python app.py Abra no celular: http://IP-DO-SEU-PC:5000 (mesmo Wi-Fi) """ import io import random import secrets import sqlite3 from datetime import date, datetime, timedelta import qrcode from flask import (Flask, abort, g, redirect, render_template_string, request, send_file) DB = "disciplina.db" app = Flask(__name__) PARABENS = [ "Parabéns! Mais um passo na direção certa! 🎉", "Mandou bem! Disciplina é isso! 💪", "Tarefa concluída! Orgulho de você! 🏆", "Constância vence talento. Continue! 🔥", "Excelente! Missão cumprida por hoje! ⭐", ] # ---------------------------------------------------------------- banco def db(): if "db" not in g: g.db = sqlite3.connect(DB) g.db.row_factory = sqlite3.Row return g.db @app.teardown_appcontext def fechar(_): d = g.pop("db", None) if d: d.close() def init_db(): con = sqlite3.connect(DB) con.executescript(""" CREATE TABLE IF NOT EXISTS tarefas( id INTEGER PRIMARY KEY AUTOINCREMENT, nome TEXT NOT NULL, tipo TEXT NOT NULL, -- distancia | leitura | habito meta REAL NOT NULL DEFAULT 1, token TEXT UNIQUE NOT NULL, inicio TEXT -- cronômetro de leitura em andamento ); CREATE TABLE IF NOT EXISTS registros( id INTEGER PRIMARY KEY AUTOINCREMENT, tarefa_id INTEGER NOT NULL, dia TEXT NOT NULL, valor REAL NOT NULL, criado TEXT NOT NULL ); """) if con.execute("SELECT COUNT(*) FROM tarefas").fetchone()[0] == 0: padrao = [("Corrida / caminhada", "distancia", 5), ("Leitura", "leitura", 30), ("Comi saudável hoje", "habito", 1), ("Beber 2L de água", "habito", 1)] for n, t, m in padrao: con.execute("INSERT INTO tarefas(nome,tipo,meta,token) VALUES(?,?,?,?)", (n, t, m, secrets.token_urlsafe(8))) con.commit() con.close() # ---------------------------------------------------------------- lógica UNIDADE = {"distancia": "km", "leitura": "min", "habito": ""} def registrar(tarefa_id, valor): db().execute("INSERT INTO registros(tarefa_id,dia,valor,criado) VALUES(?,?,?,?)", (tarefa_id, date.today().isoformat(), valor, datetime.now().isoformat(timespec="seconds"))) db().commit() def total_hoje(tarefa_id): r = db().execute("SELECT COALESCE(SUM(valor),0) s FROM registros " "WHERE tarefa_id=? AND dia=?", (tarefa_id, date.today().isoformat())).fetchone() return r["s"] def sequencia(): dias = {r["dia"] for r in db().execute("SELECT DISTINCT dia FROM registros")} d, n = date.today(), 0 if d.isoformat() not in dias: # ainda dá tempo de manter a sequência hoje d -= timedelta(days=1) while d.isoformat() in dias: n += 1 d -= timedelta(days=1) return n def total_geral(tipo): r = db().execute("SELECT COALESCE(SUM(r.valor),0) s FROM registros r " "JOIN tarefas t ON t.id=r.tarefa_id WHERE t.tipo=?", (tipo,)).fetchone() return round(r["s"], 1) # ---------------------------------------------------------------- HTML LAYOUT = """ Disciplina{{ body|safe }}""" HOME = """

🎯 Disciplina

🏃

📖
{{ lei }}
min lidos
{% for t in tarefas %}
{{ t.nome }} {% if t.feito %} ✔ concluída{% endif %}
{{ t.hoje|round(1) }} / {{ t.meta|round(1) }} {{ t.un }} hoje
NFC / QR code

Link para gravar na tag NFC:
{{ base }}t/{{ t.token }}

apagar tarefa

{% endfor %}
Nova tarefa
""" RESULTADO = """
{{ msg }}

{{ extra }}
{% if form %}
{% endif %}

← voltar

""" def pagina(corpo, **ctx): return render_template_string(LAYOUT, body=render_template_string(corpo, **ctx)) # ---------------------------------------------------------------- rotas @app.route("/") def home(): tarefas = [] for t in db().execute("SELECT * FROM tarefas ORDER BY id"): hoje = total_hoje(t["id"]) tarefas.append({**dict(t), "hoje": hoje, "un": UNIDADE[t["tipo"]], "pct": min(100, hoje / t["meta"] * 100 if t["meta"] else 0), "feito": hoje >= t["meta"]}) return pagina(HOME, tarefas=tarefas, seq=sequencia(), km=total_geral("distancia"), lei=total_geral("leitura"), base=request.host_url) @app.post("/nova") def nova(): tipo = request.form["tipo"] if tipo not in UNIDADE: abort(400) db().execute("INSERT INTO tarefas(nome,tipo,meta,token) VALUES(?,?,?,?)", (request.form["nome"].strip(), tipo, float(request.form.get("meta") or 1), secrets.token_urlsafe(8))) db().commit() return redirect("/") @app.route("/apagar/") def apagar(tid): db().execute("DELETE FROM registros WHERE tarefa_id=?", (tid,)) db().execute("DELETE FROM tarefas WHERE id=?", (tid,)) db().commit() return redirect("/") @app.route("/t/", methods=["GET", "POST"]) def tocar(token): """Endpoint chamado ao aproximar a tag NFC ou escanear o QR.""" t = db().execute("SELECT * FROM tarefas WHERE token=?", (token,)).fetchone() if not t: abort(404) un = UNIDADE[t["tipo"]] extra = f"🔥 Sequência: {sequencia()} dia(s)" if t["tipo"] == "habito": if total_hoje(t["id"]) >= t["meta"]: return pagina(RESULTADO, msg=f"✔ “{t['nome']}” já foi concluída hoje!", extra=extra) registrar(t["id"], 1) return pagina(RESULTADO, msg=f"{random.choice(PARABENS)}
“{t['nome']}”", extra=f"🔥 Sequência: {sequencia()} dia(s)") if t["tipo"] == "leitura": if not t["inicio"]: # 1ª aproximação: inicia o cronômetro db().execute("UPDATE tarefas SET inicio=? WHERE id=?", (datetime.now().isoformat(), t["id"])) db().commit() return pagina(RESULTADO, msg="📖 Leitura iniciada!
Aproxime a tag de novo ao terminar.", extra="Bom foco!") minutos = round((datetime.now() - datetime.fromisoformat(t["inicio"])).total_seconds() / 60, 1) db().execute("UPDATE tarefas SET inicio=NULL WHERE id=?", (t["id"],)) registrar(t["id"], minutos) return pagina(RESULTADO, msg=f"{random.choice(PARABENS)}
Você leu {minutos} min.", extra=f"Hoje: {total_hoje(t['id']):.1f}/{t['meta']:.0f} min · 🔥 {sequencia()} dia(s)") # distância: pede o valor (já vem preenchido com a meta) if request.method == "POST": km = float(request.form["valor"]) registrar(t["id"], km) return pagina(RESULTADO, msg=f"{random.choice(PARABENS)}
+{km:g} km registrados!", extra=f"Hoje: {total_hoje(t['id']):.1f}/{t['meta']:.0f} km · 🔥 {sequencia()} dia(s)") return pagina(RESULTADO, msg=f"🏃 {t['nome']}
Quantos km você fez?", extra="", form=True, meta=t["meta"], un=un) @app.route("/qr/.png") def qr(token): img = qrcode.make(f"{request.host_url}t/{token}") buf = io.BytesIO() img.save(buf, "PNG") buf.seek(0) return send_file(buf, mimetype="image/png") if __name__ == "__main__": init_db() app.run(host="0.0.0.0", port=5000, debug=False)
